cmake_minimum_required(VERSION 3.20)
project(TrieWise VERSION 1.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(triewise_core INTERFACE)
target_include_directories(triewise_core INTERFACE include)
if(MSVC)
  target_compile_options(triewise_core INTERFACE /W4 /permissive-)
else()
  target_compile_options(triewise_core INTERFACE -Wall -Wextra -Wpedantic)
endif()
add_executable(triewise src/main.cpp)
target_link_libraries(triewise PRIVATE triewise_core)
add_executable(triewise_tests tests/test_engine.cpp)
target_link_libraries(triewise_tests PRIVATE triewise_core)
enable_testing()
add_test(NAME engine COMMAND triewise_tests)
find_package(Python3 COMPONENTS Interpreter QUIET)
if(Python3_Interpreter_FOUND)
  add_test(NAME cli COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_cli.py
    $<TARGET_FILE:triewise> ${CMAKE_CURRENT_SOURCE_DIR}/data/dictionary.txt)
endif()
